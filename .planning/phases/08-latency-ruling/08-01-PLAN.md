```cpp
namespace k1::core::timing {

enum class EventTimeSource : std::uint8_t {
  CaptureDmaReturn = 1,
  ApPublish = 2,
  VpUse = 3,
  EmitSubmit = 4,
  TransferComplete = 5,
  UsbRead = 255  // illegal as event time
};

enum JoinMissing : std::uint32_t {
  kMissingCapture = 1u << 0,
  kMissingApPublish = 1u << 1,
  kMissingVpUse = 1u << 2,
  kMissingSubmit = 1u << 3,
  kMissingTransferComplete = 1u << 4,
  kDroppedFrame = 1u << 5,
  kEpochChange = 1u << 6,
  kUsbReadUsed = 1u << 7
};

struct AudioToLightJoin {
  std::uint64_t generation;
  std::uint64_t epoch;
  std::uint64_t newest_sample_estimate_us; // capture
  std::uint64_t ap_publish_us;
  std::uint64_t vp_use_us;                 // render_start
  std::uint64_t emit_submit_us;
  std::uint64_t emit_complete_us;          // transfer-complete
  std::uint32_t missing_mask;
};

struct AudioToLightComponents {
  std::uint32_t analysis_us;
  std::uint32_t wait_us;
  std::uint32_t render_us;
  std::uint32_t wire_us;
  std::uint32_t traced_us; // newest_sample -> emit_complete
};

bool insert_join(JoinSet& set, AudioToLightJoin record);
bool uses_usb_read_as_event_time(const AudioToLightJoin& record);
bool is_complete(const AudioToLightJoin& record);
}

Insert rules:
- UsbRead as any event-time source → reject insert, record remains with kUsbReadUsed if retained as evidence
- Dropped frame / epoch change → keep the record in the set; set kDroppedFrame / kEpochChange; do not erase
- Incomplete joins stay in the set; percentile helpers must skip them for p50 but must report n_incomplete
- Traced total uses capture and transfer-complete of the SAME generation
```

Do not include FSP/RT-Thread headers. This is shared K1 timing, not platform/.
